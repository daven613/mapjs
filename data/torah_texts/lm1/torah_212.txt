לְשׁוֹן רַבֵּנוּ זִכְרוֹנוֹ לִבְרָכָה 
זֶה בְּחִינַת מְחָאַת כַּף אֶל כַּף, כִּי עַל־יְדֵי־זֶה מַבִּיטִין בִּתְמוּנַת ה'. כִּי תְּמוּנַת ה', הַיְנוּ דִּמְיוֹנוֹת שֶׁאָנוּ מְדַמִּין אוֹתוֹ: רַחוּם וְחַנּוּן וּשְׁאָר כִּנּוּיִים וְדִמְיוֹנוֹת שֶׁאָנוּ קוֹרִין אוֹתוֹ, כָּל אֵלּוּ הַדִּמְיוֹנוֹת גִּלּוּ הַנְּבִיאִים, וּנְבִיאִים זֶה בְּחִינַת דִּבּוּר הַתְּפִלָּה, כְּמוֹ נִיב שְׂפָתַיִם, וּכְשֶׁאָנוּ מְדַבְּרִים בִּשְׂפָתֵינוּ אֶת הַתְּמוּנוֹת וְהַדִּמְיוֹנוֹת וּמָחְאִין בְּכַפַּיִם, אֲזַי נִתְקַיֵּם (הושע יב): בְּיַד הַנְּבִיאִים אֲדַמֶּה; כִּי הַנְּבִיאִים הֵם הַדִּבּוּרִים, וְהַכַּפַּיִם הֵם יַד הַנְּבִיאִים, וְאָז הַדִּמְיוֹן נִתְגַּלֶּה, בִּבְחִינַת: וּבְיַד הַנְּבִיאִים אֲדַמֶּה; וְנִתְקַיֵּם (במדבר יב): וּתְמוּנַת ה' יַבִּיט: 
גַּם מְחָאַת כַּף זֶה בְּחִינַת (יחזקאל י׳:ח׳): וִידֵי אָדָם מִתַּחַת כַּנְפֵיהֶם, כִּי הַכְּנָפַיִם הֵם הַדִּבּוּרִים, בִּבְחִינַת (קהלת י׳:כ׳): וּבַעַל כְּנָפַיִם יַגֵּיד דָּבָר: 
גַּם נִכְלָל הַתְּפִלָּה בְּתוֹךְ תּוֹרָה שֶׁבִּכְתָב וְתוֹרָה שֶׁבְּעַל־פֶּה, שֶׁהֵם שְׁנֵי יָדַיִם: יָד כּוֹתֶבֶת וְיָד חוֹתֶמֶת: 
כֹּחַ הַדִּבּוּר בָּא עַל־יְדֵי הִתְקַשְּׁרוּת לְצַדִּיקִים, כַּמּוּבָא: פֶּה אִתְקְרִיאַת מִסִּטְרָא דְּצַדִּיק: 
(מִסִּימָן ר"ט עַד כָּאן, לְשׁוֹן רַבֵּנוּ ז"ל)
